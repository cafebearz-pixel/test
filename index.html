const SHEET_DATA = 'Data'; const SHEET_CONTACT = 'Contacts'; function doGet(e) { const ss = SpreadsheetApp.getActiveSpreadsheet(); // ดึงข้อมูลคิวงาน const sheetData = ss.getSheetByName(SHEET_DATA); const dataVals = sheetData.getDataRange().getDisplayValues(); const headers = dataVals[0]; const jobs = dataVals.slice(1).map(row => { let obj = {}; headers.forEach((h, i) => obj[h] = row[i]); return obj; }); // ดึงข้อมูลเบอร์ติดต่อแบบใหม่ (6 คอลัมน์) let contacts = []; const sheetContact = ss.getSheetByName(SHEET_CONTACT); if (sheetContact) { const contactVals = sheetContact.getDataRange().getDisplayValues(); if (contactVals.length > 0) { const contactHeaders = contactVals[0]; // ตรวจสอบว่าหัวชีทเป็นเวอร์ชันใหม่หรือยัง if (contactHeaders[0] === 'ID' || contactHeaders[1] === 'ชื่อ/หน่วยงาน') { contacts = contactVals.slice(1).map(row => { let obj = {}; contactHeaders.forEach((h, i) => obj[h] = row[i]); return { id: obj['ID'] || '', fullname: obj['ชื่อ/หน่วยงาน'] || '', nickname: obj['ชื่อเล่น'] || '', position: obj['ตำแหน่ง'] || '', dept: obj['กอง'] || '', phone: obj['เบอร์โทร'] || '' }; }); } else { // กรณีหัวชีทยังเป็นเวอร์ชันเก่า contacts = contactVals.slice(1).map(r => ({ dept: r[0], phone: r[1] })); } } } return ContentService.createTextOutput(JSON.stringify({ jobs: jobs, contacts: contacts })) .setMimeType(ContentService.MimeType.JSON); } function doPost(e) { const ss = SpreadsheetApp.getActiveSpreadsheet(); const data = JSON.parse(e.postData.contents); const sheet = ss.getSheetByName(SHEET_DATA); if (data.action === "add" || data.action === "edit") { const timestamp = new Date(); const formattedReqDate = data.request_date ? new Date(data.request_date).toLocaleDateString('en-CA') : ''; const formattedActDate = data.action_date ? new Date(data.action_date).toLocaleDateString('en-CA') : ''; if (data.action === "add") { const newId = "Q-" + timestamp.getTime().toString().slice(-6); sheet.appendRow([ newId, data.task_type, formattedReqDate, data.village, data.problem_details, data.location, "ปกติ", data.requester, data.department, formattedActDate, "รับเรื่อง", "", timestamp, timestamp ]); return ContentService.createTextOutput(JSON.stringify({"status": "success"})).setMimeType(ContentService.MimeType.JSON); } else if (data.action === "edit") { const rows = sheet.getDataRange().getValues(); for (let i = 1; i < rows.length; i++) { if (rows[i][0] == data.id) { sheet.getRange(i + 1, 2, 1, 9).setValues([[ data.task_type, formattedReqDate, data.village, data.problem_details, data.location, "ปกติ", data.requester, data.department, formattedActDate ]]); sheet.getRange(i + 1, 14).setValue(timestamp); return ContentService.createTextOutput(JSON.stringify({"status": "success"})).setMimeType(ContentService.MimeType.JSON); } } } } if (data.action === "update_status") { const rows = sheet.getDataRange().getValues(); for (let i = 1; i < rows.length; i++) { if (rows[i][0] == data.id) { sheet.getRange(i + 1, 11).setValue(data.status); if (data.delay_reason !== undefined) sheet.getRange(i + 1, 12).setValue(data.delay_reason); if (data.action_date) sheet.getRange(i + 1, 10).setValue(data.action_date); sheet.getRange(i + 1, 14).setValue(new Date()); return ContentService.createTextOutput(JSON.stringify({"status": "success"})).setMimeType(ContentService.MimeType.JSON); } } } if (data.action === "update_contacts") { const sheetC = ss.getSheetByName(SHEET_CONTACT); sheetC.clear(); // สร้างหัวชีท 6 คอลัมน์ใหม่ทันที sheetC.appendRow(["ID", "ชื่อ/หน่วยงาน", "ชื่อเล่น", "ตำแหน่ง", "กอง", "เบอร์โทร"]); data.contacts.forEach(c => { sheetC.appendRow([c.id, c.fullname, c.nickname, c.position, c.dept, c.phone]); }); return ContentService.createTextOutput(JSON.stringify({"status": "success"})).setMimeType(ContentService.MimeType.JSON); } return ContentService.createTextOutput(JSON.stringify({"status": "error"})).setMimeType(ContentService.MimeType.JSON); }
